Project Overview

What this site is

JSX W&D is a solo creative studio run by Jaskirat Singh, based in Ludhiana, Punjab, India. The site serves as both a portfolio and a company front. The portfolio half earns trust by demonstrating craft. The company half converts that trust into client enquiries.

The site itself is proof #1 of capability. It must be extraordinary. If a visitor doesn't admire the site, they won't trust the work done for clients.

Prototype Reference

The file jsx-wd-prototype-v5.html is the design reference. It is a self-contained single HTML file with every visual, animation, layout, and interaction implemented. Open it in any browser. Use it as ground truth for implementation decisions.

Important

The prototype is not production code. It uses Three.js from a CDN, inline SVGs for all artwork, and has no build pipeline. The Next.js production build re-implements everything from the prototype but with proper code splitting, font optimisation, and component architecture.

Lenis Smooth Scroll

Lenis implements virtual scroll with lerp interpolation — the "gummy" quality of Locomotive's scroll. The page content moves via transform: translateY() at 7.5% of the remaining gap to target per frame. This means the scroll never snaps — it always eases in from the very first pixel.

The Core Config

The single most important value is lerp: 0.075. This is what gives the Locomotive-style gummy feel. Lower = more floaty, higher = snappier.

/* src/providers/LenisProvider.tsx */
'use client'
import { useEffect } from 'react'
import Lenis from 'lenis'

export default function LenisProvider({ children }) {
  useEffect(() => {
    const lenis = new Lenis({
      lerp: 0.075,             // ← THE KEY VALUE — Locomotive's setting
      duration: 1.2,           // momentum duration in seconds
      easing: (t) => t === 1 ? 1 : 1 - Math.pow(2, -10 * t), // expo ease-out
      smoothWheel: true,
      wheelMultiplier: 1.0,
      touchMultiplier: 0,      // let native iOS/Android momentum handle touch
      smoothTouch: false,     // never override native touch scroll
    })

    function raf(time: number) {
      lenis.raf(time)
      requestAnimationFrame(raf)
    }
    requestAnimationFrame(raf)

    // Pause when modal is open
    window.addEventListener('modal:open', () => lenis.stop())
    window.addEventListener('modal:close', () => lenis.start())

    return () => lenis.destroy()
  }, [])

  return <>{children}</>
}

Wrap your root layout:

// src/app/layout.tsx

export default function RootLayout({ children }) {
  return (
    <html>
      <body>
        <LenisProvider>{children}</LenisProvider>

  )
}

Scroll Reveal — useInView hook
// src/hooks/useInView.ts
import { useEffect, useRef, useState } from 'react'

export function useInView(threshold = 0.1) {
  const ref = useRef<HTMLElement>(null)
  const [inView, setInView] = useState(false)

  useEffect(() => {
    const el = ref.current
    if (!el) return

    const observer = new IntersectionObserver(
      ([entry]) => { if (entry.isIntersecting) setInView(true) },
      { threshold, rootMargin: '0px 0px -30px 0px' }
    )
    observer.observe(el)
    return () => observer.disconnect()
  }, [threshold])

  return { ref, inView }
}

// Usage in any section component
const { ref, inView } = useInView()

<motion.div
  ref={ref}
  initial={{ opacity: 0, y: 40 }}
  animate={inView ? { opacity: 1, y: 0 } : {}}
  transition={{ duration: 0.85, ease: [.34,1.56,.64,1] }} // spring
>
  content
</motion.div>

Grain Overlay

The prototype uses a canvas that regenerates every frame at 60fps — fully animated, flickering grain (like real 35mm film). This is expensive. For production use one of two approaches:

Option A — Throttled canvas (keeps the flicker, cheaper):

let frameCount = 0
function drawGrain() {
  frameCount++
  if (frameCount % 3 === 0) { // 20fps instead of 60fps — visually identical
    const imageData = ctx.createImageData(W, H)
    const data = imageData.data
    for (let i = 0; i < data.length; i += 4) {
      const v = Math.random() * 255 | 0
      data[i] = data[i+1] = data[i+2] = v
      data[i+3] = 255
    }
    ctx.putImageData(imageData, 0, 0)
  }
  requestAnimationFrame(drawGrain)
}

Option B — SVG filter (zero CPU, static texture):

/* In globals.css */
#grain-filter {
  position: fixed; inset: 0; z-index: 9000;
  pointer-events: none; opacity: .04; mix-blend-mode: screen;
  filter: url(#noise);
}

/* In layout — add this SVG once */
<svg style="display:none">
  <filter id="noise">
    <feTurbulence type="fractalNoise" baseFrequency="0.85"
      numOctaves="4" stitchTiles="stitch"/>
    <feColorMatrix type="saturate" values="0"/>
  </filter>
</svg>
<div id="grain-filter"></div>

Use Option A if you want the authentic Locomotive-style live grain. Use Option B if Lighthouse score is the priority.

Custom Cursor
// components/ui/Cursor.tsx — simplified production version
'use client'
import { useEffect, useRef } from 'react'

export function Cursor() {
  const dotRef = useRef<HTMLDivElement>(null)
  const ringRef = useRef<HTMLDivElement>(null)
  const labelRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const isTouch = 'ontouchstart' in window
    if (isTouch) return // no cursor on touch devices

    let mx = 0, my = 0, rx = 0, ry = 0

    document.addEventListener('mousemove', e => {
      mx = e.clientX; my = e.clientY
      if (dotRef.current) {
        dotRef.current.style.left = mx + 'px'
        dotRef.current.style.top = my + 'px'
      }
    }, { passive: true })

    function lerp() {
      rx += (mx - rx) * 0.11
      ry += (my - ry) * 0.11
      if (ringRef.current) {
        ringRef.current.style.left = rx + 'px'
        ringRef.current.style.top = ry + 'px'
      }
      requestAnimationFrame(lerp)
    }
    requestAnimationFrame(lerp)
  }, [])

  return (
    <>
      <div ref={dotRef} className="cursor-dot" />
      <div ref={ringRef} className="cursor-ring" />
      <div ref={labelRef} className="cursor-label" />
    </>
  )
}

// Headline scroll-driven font weight — run on Lenis scroll event
function updateFontWeight(scrollY: number) {
  const heroHeight = document.getElementById('hero')?.offsetHeight ?? innerHeight
  const progress = Math.min(Math.max(scrollY / heroHeight, 0), 1)
  const weight = Math.round(900 - progress * 800)   // 900 → 100
  headline.style.fontVariationSettings = `'wght' ${weight}`
  headline.style.letterSpacing = `${-0.01 + progress * 0.04}em`
}

Resend + React Email

Contact form sends two emails on submit: a notification to Jaskirat and a branded "Thank you" auto-reply to the client.

// src/app/api/contact/route.ts
import { Resend } from 'resend'
import { ThankYouEmail } from '@/emails/ThankYou'

const resend = new Resend(process.env.RESEND_API_KEY)

export async function POST(req: Request) {
  const { name, email, type, message } = await req.json()

  // 1. Notify Jaskirat
  await resend.emails.send({
    from: 'noreply@jsx-wd.com',
    to: 'hello@jsx-wd.com',
    subject: `New enquiry: ${type} from ${name}`,
    text: `From: ${name} (${email})\nType: ${type}\n\n${message}`,
  })

  // 2. Auto-reply to client
  await resend.emails.send({
    from: 'Jaskirat @ JSX W&D <hello@jsx-wd.com>',
    to: email,
    subject: 'Got your brief.',
    react: <ThankYouEmail name={name} type={type} />,
  })

  // 3. Log to Supabase
  await supabase.from('contacts').insert({ name, email, type, message })

  return Response.json({ ok: true })
}
